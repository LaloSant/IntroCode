INICIO
	
	IMPRIMIR "SISTEMADECALIFICACIONES" ; 
	#DECLARACION DE VARIABLES
	DECLARAR CADENA alumno = "CARLOS,FERNANDO;" ;
	DECLARAR ENTERO edad = 18.9 ;
	DECLARAR REAL calificacion = 9.5 ;
	DECLARAR REAL suma =  9 ;
	#DECLARAR FLOTANTE float = 0 ;
	DECLARAR ENTERO contador = s ;
	DECLARAR BOOLEANO aprobado = FALSO ;
	DECLARAR CADENA c = "cadena" ;

	#MENSAJE INICIAL
	
	IMPRIMIR "Bienvenido" , ENTER , "Al-sistema" ;

	#MOSTRAR DATOS
	IMPRIMIR "ALUMNO" , alumno , ENTER ;
	IMPRIMIR "EDAD" , edad , ENTER ;

	#LECTURA DE CALIFICACION
	LEER EN calificaci0n ;

	#VALIDACION DE CALIFICACION
	SI ( ( calificacion >= 7 ) ) ENTONCES

		SOBREESCRIBIR aprobado = VERDADERO ;

		IMPRIMIR "ESTADO->APROBADO" ;
		IMPRIMIR ENTER ;

	SINO ENTONCES

		SOBREESCRIBIR aprobado = FALSO ;

		IMPRIMIR "ESTADO->REPROBADO" ;
		IMPRIMIR ENTER ;

	FINSINO ;
	FINSI ;

	#CICLO MIENTRAS
	MIENTRAS ( contador < 4 ) HACER

		IMPRIMIR "CONTADOR" , contador , ENTER ;

		SOBREESCRIBIR contador = contador + 1 ;

	FINMIENTRAS ;

	#CICLO PARA
	PARA ( DECLARAR ENTERO i = 1 ; ) HASTA ( i <= 5 ) INCREMENTA ( 1 ) HACER

		IMPRIMIR "ITERACION" , i , ENTER ;
	FINPARA ;

	#OPERACIONES
	SOBREESCRIBIR suma = calificacion + 5 ;

	IMPRIMIR "RESULTADO-FINAL" , suma , ENTER ;

	#CONDICIONAL ANIDADO
	SI ( aprobado == VERDADERO ) ENTONCES

		SI ( suma >= 10 ) ENTONCES

			IMPRIMIR "EXCELENTE-RESULTADO" ;

		SINO ENTONCES
			PARA ( DECLARAR ENTERO i = 1 ; ) HASTA ( i <= 5 ) INCREMENTA ( 1 ) HACER
				IMPRIMIR "ITERACION" , i , ENTER ;
			FINPARA ;

			IMPRIMIR "RESULTADO-NORMAL" ;

		FINSINO ;
		FINSI ;

	FINSI ;

	#FIN DEL PROGRAMA
	IMPRIMIR ENTER ;
	IMPRIMIR "FIN-DEL-SISTEMA" ;

	DECLARAR CADENA alumnoDos = "Eduardo,bautista" ;
	DECLARAR ENTERO edadTres = 18.9 ;
	DECLARAR REAL calificacionDos = 9.5 ;
	DECLARAR REAL sumaTotal =  9 ;

	SI ( ( calificacionDos >= 7 ) ) ENTONCES
		IMPRIMIR "ESTADO->APROBADO" ;
		IMPRIMIR ENTER ;
	SINO ENTONCES
		IMPRIMIR "ESTADO->REPROBADO" ;
		IMPRIMIR ENTER ;

	FINSINO ;
	FINSI ;

FIN
