INICIO

	#DECLARACION DE VARIABLES
	DECLARAR ENTERO contador = 0 ;
	DECLARAR ENTERO limite = 5 ;
	DECLARAR REAL promedio = 8.5 ;
	DECLARAR CADENA nombre = " DANIEL " ;
	DECLARAR CADENA mensaje = " RESULTADO FINAL " ;
	DECLARAR BOOLEANO activo = VERDADERO ;

	#IMPRESION DE DATOS
	IMPRIMIR " INICIO DEL PROGRAMA " ;
	IMPRIMIR ENTER ;
	IMPRIMIR nombre , ENTER ;
	IMPRIMIR mensaje , ENTER ;

	#LECTURA DE DATOS
	LEER EN contador ;

	#CONDICIONAL SIMPLE
	SI ( contador > 0 ) ENTONCES
		IMPRIMIR " NUMERO POSITIVO " ;
	FINSI ;

	#CONDICIONAL CON SINO
	SI ( contador == limite ) ENTONCES
		IMPRIMIR " VALOR IGUAL AL LIMITE " ;
	SINO ENTONCES
		IMPRIMIR " VALOR DIFERENTE DEL LIMITE " ;
	FINSINO ;
	FINSI ;

	#CICLO MIENTRAS
	MIENTRAS ( contador < limite ) HACER
		IMPRIMIR contador ;
		SOBREESCRIBIR contador = contador + 1 ;
	FINMIENTRAS ;

	#CICLO PARA CON INCREMENTO
	PARA ( DECLARAR ENTERO i = 1 ) HASTA ( i < 4 ) INCREMENTA ( 1 ) HACER
		IMPRIMIR i ;
	FINPARA ;

	#CICLO PARA CON DECREMENTO
	PARA ( DECLARAR ENTERO j = 5 ) HASTA ( j > 0 ) DECREMENTA ( 1 ) HACER
		IMPRIMIR j ;
	FINPARA ;

	#CONDICIONALES ANIDADOS
	SI ( activo == VERDADERO ) ENTONCES

		SI ( promedio >= 7 ) ENTONCES
			IMPRIMIR " APROBADO " ;
		SINO ENTONCES
			IMPRIMIR " REPROBADO " ;
		FINSINO ;
		FINSI ;

	FINSI ;

	#FIN DEL PROGRAMA
	IMPRIMIR " PROGRAMA FINALIZADO " ;
	IMPRIMIR ENTER ;

FIN
